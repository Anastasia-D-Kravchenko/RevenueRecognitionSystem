STEP 12 — RENEW SUBSCRIPTION 1
----------------------------------------------------------------

    Server:   localhost,1433
    User:     SA
    Password: YourStrong!Passw0rd
    Database: RevenueRecognition

Test it with:  SELECT 1;


[ERROR] Right amount but current period already paid
----------------------------------------------------------------
POST /api/subscriptions/1/payments

    { "amount": 190.00 }

Expected: 409 Conflict

    { "message": "Current renewal period has already been paid." }


>>> BACKDATE THE SUBSCRIPTION <<<

USE RevenueRecognition;

UPDATE Subscription
SET StartDate        = DATEADD(MONTH, -1, GETUTCDATE()),
    CurrentPeriodEnd = DATEADD(MINUTE, -1, GETUTCDATE())
WHERE SubscriptionId = 1;

UPDATE SubscriptionPayment
SET PeriodStart = DATEADD(MONTH, -1, GETUTCDATE()),
    PeriodEnd   = DATEADD(MINUTE, -1, GETUTCDATE()),
    PaidAt      = DATEADD(MONTH, -1, GETUTCDATE())
WHERE SubscriptionId = 1;


[ERROR] Wrong amount (after backdate, we're in period 2 unpaid)
----------------------------------------------------------------
POST /api/subscriptions/1/payments

    { "amount": 200.00 }

Expected: 400 Bad Request

    { "message": "Renewal payment must equal 190 (loyal customer discount applied)." }


[HAPPY PATH] Right amount — period 2 accepted
----------------------------------------------------------------
POST /api/subscriptions/1/payments

    { "amount": 190.00 }

Expected: 200 OK

    {
      "subscriptionId":      1,
      "status":              "Active",
      "currentPeriodEnd":    "<advanced by 1 month>"
    }


[ERROR] Pay again immediately — period 2 just paid
----------------------------------------------------------------
POST /api/subscriptions/1/payments

    { "amount": 190.00 }

Expected: 409 Conflict

    { "message": "Current renewal period has already been paid." }


>>> SECOND BACKDATE <<<

USE RevenueRecognition;

DELETE FROM SubscriptionPayment
WHERE SubscriptionId = 1
  AND PeriodStart > DATEADD(MONTH, -3, GETUTCDATE());

UPDATE Subscription
SET StartDate        = DATEADD(MONTH, -3, GETUTCDATE()),
    CurrentPeriodEnd = DATEADD(MONTH, -2, GETUTCDATE())
WHERE SubscriptionId = 1;

UPDATE SubscriptionPayment
SET PeriodStart = DATEADD(MONTH, -3, GETUTCDATE()),
    PeriodEnd   = DATEADD(MONTH, -2, GETUTCDATE()),
    PaidAt      = DATEADD(MONTH, -3, GETUTCDATE())
WHERE SubscriptionId = 1;


[ERROR] Overdue — periods 2 and 3 were never paid → cancels
----------------------------------------------------------------
POST /api/subscriptions/1/payments

    { "amount": 190.00 }

Expected: 409 Conflict

    { "message": "Subscription is overdue (a previous renewal period was not paid). It has been cancelled." }


[ERROR] Try to pay a cancelled subscription
----------------------------------------------------------------

    { "amount": 190.00 }

Expected: 409 Conflict

    { "message": "Subscription is cancelled." }
